local Dragging = false
local DragInput
local DragStart
local StartPosition

local function UpdateDrag(Input)

    local Delta =
        Input.Position - DragStart

    Main.Position =
        UDim2.new(
            StartPosition.X.Scale,
            StartPosition.X.Offset + Delta.X,

            StartPosition.Y.Scale,
            StartPosition.Y.Offset + Delta.Y
        )
end


Header.InputBegan:Connect(function(Input)

    if Input.UserInputType ==
        Enum.UserInputType.MouseButton1
        or
        Input.UserInputType ==
        Enum.UserInputType.Touch then

        Dragging = true
        DragStart = Input.Position
        StartPosition = Main.Position

        DragInput = Input

        Input.Changed:Connect(function()

            if Input.UserInputState ==
                Enum.UserInputState.End then

                Dragging = false
                DragInput = nil
            end

        end)
    end
end)


Header.InputChanged:Connect(function(Input)

    if Input.UserInputType ==
        Enum.UserInputType.MouseMovement
        or
        Input.UserInputType ==
        Enum.UserInputType.Touch then

        DragInput = Input
    end
end)


UserInputService.InputChanged:Connect(function(Input)

    if not Dragging then
        return
    end

    if Input == DragInput then
        UpdateDrag(Input)
    end
end)


--//====================================================
--// DRAG DA BOLINHA - PC + MOBILE
--//====================================================

local BallDragging = false
local BallDragStart
local BallStartPosition
local BallDragInput
local BallMoved = false

local function UpdateBallDrag(Input)
    local Delta = Input.Position - BallDragStart
    if Delta.Magnitude > 5 then
        BallMoved = true
    end

    OpenButton.Position = UDim2.new(
        BallStartPosition.X.Scale,
        BallStartPosition.X.Offset + Delta.X,
        BallStartPosition.Y.Scale,
        BallStartPosition.Y.Offset + Delta.Y
    )
end

OpenButton.InputBegan:Connect(function(Input)
    if Input.UserInputType == Enum.UserInputType.MouseButton1
        or Input.UserInputType == Enum.UserInputType.Touch then

        BallDragging = true
        BallMoved = false
        BallDragStart = Input.Position
        BallStartPosition = OpenButton.Position
        BallDragInput = Input

        TweenService:Create(BallScale, TweenInfo.new(0.1, Enum.EasingStyle.Quad), {
            Scale = 0.92
        }):Play()

        Input.Changed:Connect(function()
            if Input.UserInputState == Enum.UserInputState.End then
                BallDragging = false
                BallDragInput = nil

                TweenService:Create(BallScale, TweenInfo.new(0.18, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
                    Scale = 1
                }):Play()
            end
        end)
    end
end)

OpenButton.InputChanged:Connect(function(Input)
    if Input.UserInputType == Enum.UserInputType.MouseMovement
        or Input.UserInputType == Enum.UserInputType.Touch then
        BallDragInput = Input
    end
end)

UserInputService.InputChanged:Connect(function(Input)
    if BallDragging and Input == BallDragInput then
        UpdateBallDrag(Input)
    end
end)

local InterfaceOpen = true

local function OpenInterface()

    InterfaceOpen = true

    Main.Visible = true
    MainScale.Scale = 0.78

    Main.BackgroundTransparency = 0.25

    local ScaleTween =
        TweenService:Create(
            MainScale,

            TweenInfo.new(
                0.3,
                Enum.EasingStyle.Back,
                Enum.EasingDirection.Out
            ),

            {
                Scale = 1
            }
        )

    local FadeTween =
        TweenService:Create(
            Main,

            TweenInfo.new(0.2),

            {
                BackgroundTransparency = 0
            }
        )

    ScaleTween:Play()
    FadeTween:Play()
end


local function CloseInterface()

    InterfaceOpen = false

    local ScaleTween =
        TweenService:Create(
            MainScale,

            TweenInfo.new(
                0.22,
                Enum.EasingStyle.Quad,
                Enum.EasingDirection.In
            ),

            {
                Scale = 0.78
            }
        )

    local FadeTween =
        TweenService:Create(
            Main,

            TweenInfo.new(0.18),

            {
                BackgroundTransparency = 0.25
            }
        )

    ScaleTween:Play()
    FadeTween:Play()

    ScaleTween.Completed:Wait()

    if not InterfaceOpen then
        Main.Visible = false
    end
end


OpenButton.MouseButton1Click:Connect(function()
    if BallMoved then
        BallMoved = false
        return
    end

    if InterfaceOpen then
        CloseInterface()
    else
        OpenInterface()
    end
end)

local function CreateSlider(
    Y,
    Name,
    Min,
    Max,
    Default,
    Callback
)

    local Container = Instance.new("Frame")

    Container.Size =
        UDim2.new(1,-28,0,57)

    Container.Position =
        UDim2.fromOffset(14,Y)

    Container.BackgroundColor3 = Color3.fromRGB(19,19,23)
    Container.BackgroundTransparency = 0
    Container.BorderSizePixel = 0
    Container.Parent = Main

    local ContainerCorner = Instance.new("UICorner")
    ContainerCorner.CornerRadius = UDim.new(0,10)
    ContainerCorner.Parent = Container

    local ContainerStroke = Instance.new("UIStroke")
    ContainerStroke.Thickness = 1
    ContainerStroke.Transparency = 0.78
    ContainerStroke.Parent = Container

    --// LABEL
    local Label = Instance.new("TextLabel")

    Label.Size =
        UDim2.new(1,-22,0,20)

    Label.Position =
        UDim2.fromOffset(11,2)

    Label.BackgroundTransparency = 1

    Label.Text =
        Name .. ": " .. Default

    Label.TextColor3 =
        Color3.fromRGB(225,225,230)

    Label.TextSize = 11
    Label.Font = Enum.Font.GothamMedium

    Label.TextXAlignment =
        Enum.TextXAlignment.Left

    Label.Parent = Container


    --// BAR
    local Bar = Instance.new("Frame")

    Bar.Size =
        UDim2.new(1,-22,0,7)

    Bar.Position =
        UDim2.fromOffset(11,31)

    Bar.BackgroundColor3 =
        Color3.fromRGB(38,38,44)

    Bar.BorderSizePixel = 0
    Bar.Active = true
    Bar.Parent = Container

    local BarCorner = Instance.new("UICorner")
    BarCorner.CornerRadius = UDim.new(1,0)
    BarCorner.Parent = Bar


    --// FILL
    local Fill = Instance.new("Frame")

    Fill.BackgroundColor3 =
        Color3.fromRGB(225,55,65)

    Fill.BorderSizePixel = 0
    Fill.Parent = Bar

    local FillCorner = Instance.new("UICorner")
    FillCorner.CornerRadius = UDim.new(1,0)
    FillCorner.Parent = Fill


    --// KNOB
    local Knob = Instance.new("TextButton")

    Knob.Size =
        UDim2.fromOffset(18,18)

    Knob.AnchorPoint =
        Vector2.new(0.5,0.5)

    Knob.BackgroundColor3 =
        Color3.fromRGB(255,255,255)

    Knob.BorderSizePixel = 0
    Knob.Text = ""
    Knob.AutoButtonColor = false
    Knob.ZIndex = 5
    Knob.Parent = Bar

    local KnobCorner = Instance.new("UICorner")
    KnobCorner.CornerRadius = UDim.new(1,0)
    KnobCorner.Parent = Knob

    local KnobScale = Instance.new("UIScale")
    KnobScale.Scale = 1
    KnobScale.Parent = Knob


    local Sliding = false


    --//================================================
    --// APLICA VALOR
    --//================================================

    local function ApplyValue(Value)

        if not Humanoid then
            Callback(Value)
            return
        end


        if Name == "SPEED" then

            -- Guarda o valor atual antes da alteração
            local OldValue =
                Humanoid.WalkSpeed

            -- Restaura primeiro
            Humanoid.WalkSpeed =
                OldValue

            task.wait()

            -- Aplica o novo valor
            Humanoid.WalkSpeed =
                Value


        elseif Name == "JUMP" then

            if Humanoid.UseJumpPower then

                local OldValue =
                    Humanoid.JumpPower

                Humanoid.JumpPower =
                    OldValue

                task.wait()

                Humanoid.JumpPower =
                    Value

            else

                local OldValue =
                    Humanoid.JumpHeight

                Humanoid.JumpHeight =
                    OldValue

                task.wait()

                Humanoid.JumpHeight =
                    Value
            end
        end

        Callback(Value)
    end


    --//================================================
    --// SET VALUE
    --//================================================

    local function SetValue(X)

        local Width =
            Bar.AbsoluteSize.X

        if Width <= 0 then
            return
        end

        local Percent =
            math.clamp(
                (X - Bar.AbsolutePosition.X) / Width,
                0,
                1
            )

        local Value =
            math.floor(
                Min +
                ((Max-Min) * Percent)
                + 0.5
            )


        Label.Text =
            Name .. ": " .. Value


        TweenService:Create(
            Fill,

            TweenInfo.new(
                0.07,
                Enum.EasingStyle.Quad,
                Enum.EasingDirection.Out
            ),

            {
                Size =
                    UDim2.new(
                        Percent,
                        0,
                        1,
                        0
                    )
            }
        ):Play()


        TweenService:Create(
            Knob,

            TweenInfo.new(
                0.07,
                Enum.EasingStyle.Quad,
                Enum.EasingDirection.Out
            ),

            {
                Position =
                    UDim2.new(
                        Percent,
                        0,
                        0.5,
                        0
                    )
            }
        ):Play()


        ApplyValue(Value)
    end


    --//================================================
    --// BEGIN SLIDE
    --//================================================

    local function BeginSlide(Input)

        if Input.UserInputType ~=
            Enum.UserInputType.MouseButton1
            and
            Input.UserInputType ~=
            Enum.UserInputType.Touch then

            return
        end

        Sliding = true

        TweenService:Create(
            KnobScale,

            TweenInfo.new(0.12),

            {
                Scale = 1.3
            }
        ):Play()

        SetValue(Input.Position.X)
    end


    Knob.InputBegan:Connect(BeginSlide)
    Bar.InputBegan:Connect(BeginSlide)


    --//================================================
    --// SLIDE
    --//================================================

    UserInputService.InputChanged:Connect(function(Input)

        if not Sliding then
            return
        end

        if Input.UserInputType ==
            Enum.UserInputType.MouseMovement
            or
            Input.UserInputType ==
            Enum.UserInputType.Touch then

            SetValue(Input.Position.X)
        end
    end)


    --//================================================
    --// END SLIDE
    --//================================================

    UserInputService.InputEnded:Connect(function(Input)

        if Input.UserInputType ==
            Enum.UserInputType.MouseButton1
            or
            Input.UserInputType ==
            Enum.UserInputType.Touch then

            if Sliding then

                Sliding = false

                TweenService:Create(
                    KnobScale,

                    TweenInfo.new(
                        0.16,
                        Enum.EasingStyle.Back,
                        Enum.EasingDirection.Out
                    ),

                    {
                        Scale = 1
                    }
                ):Play()
            end
        end
    end)


    --//================================================
    --// DEFAULT
    --//================================================

    task.defer(function()

        local Percent =
            (Default-Min)/(Max-Min)

        Fill.Size =
            UDim2.new(
                Percent,
                0,
                1,
                0
            )

        Knob.Position =
            UDim2.new(
                Percent,
                0,
                0.5,
                0
            )
    end)
end

--// AUTO RETURN
--//====================================================

ProximityPromptService.PromptTriggered:Connect(
    function(
        Prompt,
        PlayerWhoTriggered
    )

        if not AutoReturn then
            return
        end

        if PlayerWhoTriggered
            and
            PlayerWhoTriggered ~= Player then

            return
        end

        if Returning then
            return
        end


        local CharacterNow =
            Player.Character

        if not CharacterNow then
            return
        end


        local RootNow =
            CharacterNow:
            FindFirstChild(
                "HumanoidRootPart"
            )

        if not RootNow then
            return
        end


        Returning = true

        Status.Text =
            "●  Status: Prompt → retornando"


        local Distance =
            (
                RootNow.Position
                -
                ReturnPosition
            ).Magnitude


        local Duration =
            math.max(
                Distance / ReturnSpeed,
                0.05
            )


        local Tween =
            TweenService:Create(

                RootNow,

                TweenInfo.new(
                    Duration,
                    Enum.EasingStyle.Linear,
                    Enum.EasingDirection.Out
                ),

                {
                    CFrame =
                        CFrame.new(
                            ReturnPosition
                        )
                }
            )


        Tween:Play()

        Tween.Completed:Wait()


        Returning = false

        Status.Text =
            "●  Status: Na posição"
    end
)

task.spawn(function()

    while task.wait(0.1) do

        if Humanoid then

            -- Mantém o Speed escolhido
            -- mesmo se outro script tentar
            -- modificar o valor.

            if Humanoid.WalkSpeed ~= Speed then

                Humanoid.WalkSpeed =
                    Speed
            end


            -- Mantém o Jump escolhido.

            if Humanoid.UseJumpPower then

                if Humanoid.JumpPower ~= Jump then

                    Humanoid.JumpPower =
                        Jump
                end

            else

                if Humanoid.JumpHeight ~= Jump then

                    Humanoid.JumpHeight =
                        Jump
                end
            end
        end
    end
end)
